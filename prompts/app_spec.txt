<project_specification>
  <project_name>Student Work Archive System</project_name>

  <overview>
    A web-based system for archiving, browsing, and showcasing student works (projects, reports, theses, presentations, etc.) stored on Google Shared Drive. Users can view, vote, and comment on submissions while administrators manage content organization and system configuration. The system integrates with Google OAuth for authentication and Google Drive API for file storage and streaming.
  </overview>

  <technology_stack>
    <frontend>
      <framework>React + Vite</framework>
      <styling>Tailwind CSS with shadcn/ui components</styling>
      <pdf_viewer>PDF.js (Mozilla's PDF rendering library)</pdf_viewer>
      <charts>Recharts (for analytics dashboard)</charts>
    </frontend>
    <backend>
      <runtime>Node.js + Express</runtime>
      <database>SQLite via better-sqlite3</database>
      <authentication>Passport.js + Google OAuth 2.0</authentication>
      <email>Nodemailer (for notifications)</email>
    </backend>
    <communication>
      <api>REST API</api>
      <format>JSON with standardized response structure</format>
    </communication>
    <integrations>
      <file_storage>Google Drive API v3 (Shared Drive)</file_storage>
      <search>SQLite FTS5 (built-in full-text search)</search>
    </integrations>
  </technology_stack>

  <prerequisites>
    <environment_setup>
      - Node.js 18+ installed
      - Google Cloud Project with Drive API and OAuth 2.0 enabled
      - Google OAuth 2.0 credentials (Client ID and Secret)
      - Access to a Google Shared Drive for file storage
      - SMTP server credentials (optional, for email notifications)
    </environment_setup>
  </prerequisites>

          - /admin/users
          - /admin/analytics
        </protected_routes>

      <role name="user">
        <description>Authenticated user with full interaction capabilities</description>
        <permissions>
          - Browse and search all public works

  <database_schema>
    <tables>
      <categories>
        - id (INTEGER PRIMARY KEY)
        - name (TEXT NOT NULL)
        - description (TEXT)
        - parent_id (INTEGER, self-reference for hierarchy)
        - sort_order (INTEGER DEFAULT 0)
        - created_at (DATETIME)
      </categories>

      <work_types>
        - id (INTEGER PRIMARY KEY)
        - name (TEXT NOT NULL) - 'Project', 'Report', 'Thesis', etc.
        - icon (TEXT)
        - created_at (DATETIME)
      </work_types>

      <tags>
        - id (INTEGER PRIMARY KEY)
        - name (TEXT UNIQUE NOT NULL)
        - color (TEXT DEFAULT '#6B7280')
      </tags>

      <work_tags>
        - work_id (INTEGER, FK to works)
        - tag_id (INTEGER, FK to tags)
        - PRIMARY KEY (work_id, tag_id)
      </work_tags>

      <works>
        - id (INTEGER PRIMARY KEY)
        - title (TEXT NOT NULL)
        - description (TEXT)
        - author_name (TEXT)
        - author_email (TEXT)
        - academic_year (TEXT)
        - work_type_id (INTEGER, FK to work_types)
        - category_id (INTEGER, FK to categories)
        - google_file_id (TEXT NOT NULL)
        - file_url (TEXT)
        - thumbnail_url (TEXT)
        - file_size (INTEGER)
        - page_count (INTEGER)
        - view_count (INTEGER DEFAULT 0)
        - download_count (INTEGER DEFAULT 0)
        - is_featured (INTEGER DEFAULT 0)
        - is_public (INTEGER DEFAULT 1)
        - share_token (TEXT UNIQUE)
        - created_at, updated_at (DATETIME)
        - INDEXES: category_id, work_type_id, is_featured
        - FTS5 virtual table for search (title, description, author_name)
      </works>

      <users>
        - id (INTEGER PRIMARY KEY)
        - google_id (TEXT UNIQUE NOT NULL)
        - email (TEXT UNIQUE NOT NULL)
        - name (TEXT)
        - avatar_url (TEXT)
        - role (TEXT DEFAULT 'user', CHECK IN ('admin', 'user', 'guest'))
        - notify_new_works (INTEGER DEFAULT 0)
        - notify_comments (INTEGER DEFAULT 1)
        - last_login (DATETIME)
        - created_at (DATETIME)
      </users>

      <votes>
        - id (INTEGER PRIMARY KEY)
        - work_id (INTEGER, FK to works ON DELETE CASCADE)
        - user_id (INTEGER, FK to users ON DELETE CASCADE)
        - stars (INTEGER, CHECK 0-5)
        - created_at (DATETIME)
        - UNIQUE(work_id, user_id)
        - INDEX: work_id
      </votes>

      <comments>
        - id (INTEGER PRIMARY KEY)
        - work_id (INTEGER, FK to works ON DELETE CASCADE)
        - user_id (INTEGER, FK to users ON DELETE CASCADE)
        - parent_id (INTEGER, self-reference for replies)
        - content (TEXT NOT NULL)
        - is_deleted (INTEGER DEFAULT 0)
        - created_at, updated_at (DATETIME)
        - INDEX: work_id
      </comments>

      <favorites>
        - id (INTEGER PRIMARY KEY)
        - user_id (INTEGER, FK to users ON DELETE CASCADE)
        - work_id (INTEGER, FK to works ON DELETE CASCADE)
        - created_at (DATETIME)
        - UNIQUE(user_id, work_id)
      </favorites>

      <activity_logs>
        - id (INTEGER PRIMARY KEY)
        - work_id (INTEGER, FK to works ON DELETE CASCADE)
        - user_id (INTEGER, FK to users, nullable)
        - action (TEXT, CHECK IN ('view', 'download', 'share'))
        - ip_address (TEXT)
        - user_agent (TEXT)
        - created_at (DATETIME)
        - INDEXES: work_id, created_at
      </activity_logs>

      <config>
        - key (TEXT PRIMARY KEY)
        - value (TEXT)
        - description (TEXT)
        - updated_at (DATETIME)
        - Default entries: google_shared_drive_id, google_shared_drive_url, allowed_domain, site_name, site_description, items_per_page, allow_guest_view, require_approval, smtp_host, smtp_port, smtp_user, smtp_pass
      </config>
    </tables>
  </database_schema>

  <api_endpoints_summary>
    <authentication>

    <works>

      - GET /api/works/search?q= - Full-text search

      - POST /api/works/:id/view - Record view, increment count
      - GET /api/works/:id/download - Get download URL, increment count

      - DELETE /api/comments/:id - Delete comment

      - POST /admin/tags - Create tag

    <admin_users>
      - GET /admin/users - List users
      - PUT /admin/users/:id/role - Change user role
    </admin_users>
